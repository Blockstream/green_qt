qt_add_executable(blockstream_app)

set_target_properties(blockstream_app PROPERTIES
    AUTOMOC ON
    OUTPUT_NAME "${BLOCKSTREAM_PROJECT_NAME}"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
    QT_QML_IMPORT_SCANNER_EXTRA_ROOT_PATHS "${CMAKE_SOURCE_DIR}/qml;${CMAKE_SOURCE_DIR}/modules/theme/qml"
    LINK_SEARCH_START_STATIC ON
    LINK_SEARCH_END_STATIC ON
)

if (WIN32)
    enable_language("RC")
    set_target_properties(blockstream_app PROPERTIES
        WIN32_EXECUTABLE TRUE
    )
    target_sources(blockstream_app PRIVATE production.rc)
endif()

if(APPLE)
    set_target_properties(blockstream_app PROPERTIES
        MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in
        MACOSX_BUNDLE TRUE
        OUTPUT_NAME "${BLOCKSTREAM_DISPLAY_NAME}"
    )
    set_source_files_properties(production.icns PROPERTIES MACOSX_PACKAGE_LOCATION "Resources")
    target_sources(blockstream_app PRIVATE production.icns)
endif()

if (UNIX AND NOT APPLE AND NOT WIN32)
    target_link_options(blockstream_app PRIVATE
        -static-libgcc
        -static-libstdc++
        -fstack-protector
        -Wl,--wrap=__divmoddi4
        -Wl,--wrap=log2f
    )
    qt_import_plugins(blockstream_app
        INCLUDE Qt6::QXcbIntegrationPlugin Qt6::QWaylandIntegrationPlugin Qt6::QFFmpegMediaPlugin
    )
endif()

# blockstream_coreplugin is the QML module plugin generated by qt_add_qml_module. Linking it
# is what keeps the type registration and the QML resources of the static backing library
# from being dropped by the linker.
target_link_libraries(blockstream_app PRIVATE blockstream_core blockstream_coreplugin)

add_subdirectory(src)

install(TARGETS blockstream_app
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
